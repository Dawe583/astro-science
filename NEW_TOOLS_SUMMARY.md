## 🎯 Příklady Použití

### Kosmologie s CAMB
```python
import camb
from camb import model, initialpower

# Nastavení kosmologických parametrů
pars = camb.CAMBparams()
pars.set_cosmology(H0=67.5, ombh2=0.022, omch2=0.122, mnu=0.06)
pars.InitPower.set_params(ns=0.965, As=2e-9)

# Výpočet CMB power spectra
results = camb.get_results(pars)
powers = results.get_cmb_power_spectra(pars, CMB_unit='muK')
cl = powers['total']

# Vykreslení
import matplotlib.pyplot as plt
plt.plot(cl[0], label='TT')
plt.loglog()
plt.show()
```

### CMB Analýza s Healpy

```python
import healpy as hp
```

### Gravitational Wave Inference s Bilby
```python
import bilby
from bilby.gw.conversion import convert_to_lal_binary_black_hole_parameters

# Nastavení likelihood
likelihood = bilby.gw.likelihood.GWLikelihood(
    interferometers=['H1', 'L1'],
    waveform_generator=bilby.gw.waveform_generator.WaveformGenerator(...)
)

# Nastavení priorů
priors = bilby.gw.prior.BBHPriorDict()
priors['mass_1'] = bilby.prior.Uniform(5, 50, 'mass_1')
priors['mass_2'] = bilby.prior.Uniform(5, 50, 'mass_2')

# Spuštění sampleru
result = bilby.run_sampler(
    likelihood=likelihood,
    priors=priors,
    sampler='dynesty',
    nlive=1000
)

# Corner plot
result.plot_corner()
```

### Plasma Physics s PlasmaPy

### Atomic Simulation s ASE
```python
from ase import Atoms
from ase.calculators.emt import EMT
from ase.optimize import BFGS

# Vytvoření molekuly vody
atoms = Atoms('H2O',
              positions=[[0, 0, 0],
                        [0.95, 0, 0],
                        [0.24, 0.93, 0]])
atoms.calc = EMT()

# Výpočet energie
energy = atoms.get_potential_energy()
print(f"Potential energy: {energy} eV")

# Optimalizace geometrie
opt = BFGS(atoms)
opt.run(fmax=0.05)

# Vibrational analysis
from ase.vibrations import Vibrations
vib = Vibrations(atoms)
vib.run()
vib.summary()
```

### Quantum Computing s Qiskit
```python
from qiskit import QuantumCircuit, Aer, execute
from qiskit.visualization import plot_histogram

# Vytvoření kvantového obvodu
qc = QuantumCircuit(2, 2)
qc.h(0)  # Hadamard gate
qc.cx(0, 1)  # CNOT gate
qc.measure([0, 1], [0, 1])

# Simulace
backend = Aer.get_backend('qasm_simulator')
result = execute(qc, backend, shots=1000).result()
counts = result.get_counts()
print(counts)
```

### Quantum Machine Learning s PennyLane
```python
import pennylane as qml
from pennylane import numpy as np

# Definice kvantového zařízení
dev = qml.device('default.qubit', wires=2)

@qml.qnode(dev)
def circuit(params):
    qml.RX(params[0], wires=0)
    qml.RY(params[1], wires=1)
    qml.CNOT(wires=[0, 1])
    return qml.expval(qml.PauliZ(0))

# Optimalizace
params = np.array([0.3, 0.5])
print(circuit(params))
```
